### 3.2 Navigation Flow

**Journey: First-Time User Onboarding**
```
Landing (external)
  → Register (/register/:token)
    → Dashboard (/) [redirect after registration]
      → Project Create (/projects/new)
        → Project Overview (/projects/:projectId)
```

**Journey: File Upload and Process**
```
Dashboard (/)
  → Project Overview (/projects/:projectId)
    → Sources List (/projects/:projectId/sources)
      → Source Upload (/projects/:projectId/sources/upload)
        → Source Detail (/projects/:projectId/sources/:sourceId)
          → Schema Mapping (/projects/:projectId/sources/:sourceId/schema)
            → De-identification (/projects/:projectId/sources/:sourceId/deid)
              → Processing Config (/projects/:projectId/processing)
                → Processing Run (/projects/:projectId/processing/run)
                  → Datasets List (/projects/:projectId/datasets)
                    → Dataset Detail (/projects/:projectId/datasets/:datasetId)
                      → Export Download
```

**Journey: Connect Teamwork Desk**
```
Dashboard (/)
  → Project Overview (/projects/:projectId)
    → Sources List (/projects/:projectId/sources)
      → Source Connect Teamwork (/projects/:projectId/sources/connect/teamwork)
        → Source Detail (/projects/:projectId/sources/:sourceId)
          → [Continue with Schema Mapping flow]
```

**Journey: Team Management (Admin)**
```
Dashboard (/)
  → User Menu (dropdown)
    → Team Management (/settings/team)
      → Invite Member (modal) → sends email
      → Edit Role (modal)
```

**Journey: User Profile**
```
Dashboard (/)
  → User Menu (dropdown)
    → User Profile (/settings/profile)
      → Edit Profile (inline form)
      → Change Password (modal)
```

**Journey: Password Reset**
```
Login (/login)
  → Forgot Password (/forgot-password)
    → [email sent]
      → Reset Password (/reset-password/:token)
        → Login (/login) [redirect after reset]
```

**Journey: Organization Settings**
```
Dashboard (/)
  → User Menu (dropdown)
    → Organization Settings (/settings/organization)
      → Edit Organization (inline form)
```

### 3.6 Role-Based Gating

```tsx
// client/src/components/require-role.tsx
import { Navigate } from 'react-router-dom';
import { useAuth } from '../contexts/auth-context';

interface RequireRoleProps {
  allowedRoles: string[];
  children: React.ReactNode;
}

export function RequireRole({ allowedRoles, children }: RequireRoleProps) {
  const { user, isLoading } = useAuth();
  
  if (isLoading) return <LoadingSpinner />;
  if (!user || !allowedRoles.includes(user.role)) {
    return <Navigate to="/unauthorized" replace />;
  }
  return <>{children}</>;
}
```

**Navigation Filtering:**

```tsx
const { user } = useAuth();

const navItems = [
  { path: '/', label: 'Projects', icon: LayoutDashboard, roles: ['member', 'admin'] },

  { path: '/settings/team', label: 'Team', icon: Users, roles: ['admin'] },
];

const visibleItems = navItems.filter(item => 
  item.roles.includes(user?.role || '')
);
```

### 7.2 Global Error Handling

```typescript
// client/src/lib/api.ts

async function handleResponse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    const error = await res.json();
    
    // Handle auth errors globally
    if (error.error?.code === 'UNAUTHORIZED' || error.error?.code === 'TOKEN_EXPIRED') {
      localStorage.removeItem('auth_token');
      window.location.href = `/login?returnTo=${encodeURIComponent(window.location.pathname)}`;
      throw error;
    }
    
    throw error;
  }
  return res.json();
}
```

### 7.3 Rate Limit UI Pattern

```tsx
const [retryAfter, setRetryAfter] = useState<number | null>(null);

const handleSubmit = async (data) => {
  try {
    await api.post('/api/auth/login', data);
  } catch (error) {
    if (error.error?.code === 'RATE_LIMIT_EXCEEDED') {
      const resetTime = error.retryAfter || 60;
      setRetryAfter(resetTime);
      toast({
        variant: "destructive",
        title: "Too many attempts",
        description: `Please wait ${resetTime} seconds before trying again.`
      });
    }
  }
};

// Countdown timer
useEffect(() => {
  if (retryAfter && retryAfter > 0) {
    const timer = setTimeout(() => setRetryAfter(retryAfter - 1), 1000);
    return () => clearTimeout(timer);
  }
}, [retryAfter]);

<Button type="submit" disabled={retryAfter > 0 || isSubmitting}>
  {retryAfter > 0 ? `Wait ${retryAfter}s` : 'Sign in'}
</Button>
```
